cmake_minimum_required(VERSION 3.18.2 FATAL_ERROR)
project(omnirefl_tests LANGUAGES C CXX)
enable_testing()
get_filename_component(omnirefl_PROJECTS_DIR
    "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(NOT TARGET omni::tool)
    find_package(omnirefl CONFIG REQUIRED)
endif()

include("${omnirefl_PROJECTS_DIR}/cmake/toolchain_checks.cmake")
include("${omnirefl_PROJECTS_DIR}/cmake/test_dependencies.cmake")
set(OMNI_TEST_SUITE full CACHE STRING "Package test suite: smoke or full")
set_property(CACHE OMNI_TEST_SUITE PROPERTY STRINGS smoke full)
omnirefl_supported_cpp_standards(_supported_cpp_standards)
if(OMNI_TEST_SUITE STREQUAL "smoke" AND _supported_cpp_standards)
    list(GET _supported_cpp_standards 0 _minimum_standard)
    list(GET _supported_cpp_standards -1 _maximum_standard)
    set(_supported_cpp_standards "${_minimum_standard};${_maximum_standard}")
    list(REMOVE_DUPLICATES _supported_cpp_standards)
endif()
message(STATUS "omnirefl: ${OMNI_TEST_SUITE} test C++ standards: "
    "${_supported_cpp_standards}")
set(_standard_labels "${_supported_cpp_standards}")
list(TRANSFORM _standard_labels PREPEND "C++")
list(JOIN _standard_labels ", " _standard_labels)
file(WRITE "${CMAKE_BINARY_DIR}/test-standards.tsv"
    "${OMNI_TEST_SUITE}\t${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\t${_standard_labels}\n")
if(OMNI_TEST_SUITE STREQUAL "smoke")
    add_subdirectory(smoke)
elseif(OMNI_TEST_SUITE STREQUAL "full")
    add_subdirectory(tool)
    add_subdirectory(functional)
    add_subdirectory(ccdb_query)
else()
    message(FATAL_ERROR "OMNI_TEST_SUITE must be smoke or full")
endif()
if(NOT DEFINED BUILD_EXTENSIONS OR BUILD_EXTENSIONS)
    if(NOT TARGET omni::serialization)
        find_package(omnirefl CONFIG REQUIRED COMPONENTS serialization)
    endif()
    add_subdirectory("${omnirefl_PROJECTS_DIR}/extensions/serialization/tests"
        extensions/serialization)
endif()
