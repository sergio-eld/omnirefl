cmake_minimum_required(VERSION 3.18.2 FATAL_ERROR)

project(omnirefl_examples LANGUAGES CXX)

enable_testing()
if(NOT TARGET omni::tool)
    find_package(omnirefl CONFIG REQUIRED)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/toolchain_checks.cmake")

if(NOT "cxx_std_23" IN_LIST CMAKE_CXX_COMPILE_FEATURES)
    message(STATUS "omnirefl: skipping sneak_peek: C++23 is not available")
else()
    _omnirefl_check_cxx23_library(_examples_has_cxx23_library)
    if(_examples_has_cxx23_library)
        add_subdirectory(sneak_peek)
        if(CMAKE_SYSTEM_NAME STREQUAL CMAKE_HOST_SYSTEM_NAME)
            add_test(NAME example.sneak_peek COMMAND sneak_peek)
        endif()
    else()
        message(STATUS "omnirefl: skipping sneak_peek: C++23 library features are not available")
    endif()
endif()

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR OR TARGET omni::serialization)
    add_subdirectory(extensions/serialization/ryml)
    if(CMAKE_SYSTEM_NAME STREQUAL CMAKE_HOST_SYSTEM_NAME)
        add_test(NAME example.json_serialization COMMAND json_serialization)
    endif()
else()
    message(STATUS "omnirefl: skipping json_serialization: extensions disabled")
endif()

_omnirefl_check_cxx20_language(_examples_has_guide_features)
if(_examples_has_guide_features)
    add_subdirectory(comprehensive_guide)
else()
    message(STATUS "omnirefl: skipping comprehensive_guide: C++20 language features are not available")
endif()
